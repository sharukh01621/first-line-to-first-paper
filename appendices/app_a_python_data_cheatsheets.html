def safe_bmi(weight_kg: float, height_cm: float) -> float:
    if weight_kg <= 0 or height_cm <= 0:
        raise ValueError("weight_kg and height_cm must be positive.")
    height_m = height_cm / 100.0
    return weight_kg / (height_m ** 2)

A.2.2 Broadcasting Rules & Axis Reductions

Two array shapes are broadcast-compatible when, aligning dimensions from right to left, every dimension pair \((a, b)\) satisfies:

  1. \(a = b\), or
  2. One of \(a\) or \(b\) equals \(1\), or
  3. One array has fewer dimensions (implicitly padded with leading ones, \(1\)).
# Standardize columns: X is (n, p); col_mean/col_std are (1, p)
col_mean = np.nanmean(X, axis=0, keepdims=True)   # down rows -> (1, p)
col_std = np.nanstd(X, axis=0, keepdims=True)     # down rows -> (1, p)
X_z = (X - col_mean) / np.where(col_std == 0, 1.0, col_std)

# Row-wise L2 norm: axis=1 collapses across columns -> (n, 1)
row_norms = np.linalg.norm(X_z, ord=2, axis=1, keepdims=True)
X_unit = X_z / np.maximum(row_norms, 1e-12)

A.2.3 Linear Algebra & SVD Core Idioms

# Matrix multiply (@), transpose (.T), solve, and thin SVD
n, p = X_z.shape
cov = (X_z.T @ X_z) / (n - 1)                      # (p, p) covariance
eigvals, eigvecs = np.linalg.eigh(cov)             # symmetric matrix
U, s, Vt = np.linalg.svd(X_z, full_matrices=False) # thin SVD
explained_var_ratio = (s ** 2) / np.sum(s ** 2)

# Numerically stable Ridge solve: (X^T X + lam * I) w = X^T y
lam = 1.0
A = X_z.T @ X_z + lam * np.eye(p)
w_ridge = np.linalg.solve(A, X_z.T @ y)            # never np.linalg.inv

A.3 Pandas 3.x Copy-on-Write Wrangling (Chapters 17–18)

WarningPandas 3.x Copy-on-Write (CoW) Golden Rule

Never chain indexing on the left-hand side of an assignment (df[mask]["col"] = val). In Pandas 3.x CoW, chained assignments silently fail or raise ChainedAssignmentError. Always mutate via a single .loc[row_mask, "col"] = val call or return a new DataFrame with .assign(...).

Repeated patients, clinics, or survey clusters StratifiedGroupKFold(n_splits=5) (or GroupKFold) Same patient/cluster in train & val

A.6.2 Reproducible Dataset, DataLoader & Canonical 5-Step Loop

import random
import numpy as np
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, Dataset

def seed_everything(seed: int = 42) -> torch.device:
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    if torch.cuda.is_available():
        return torch.device("cuda")
    if torch.backends.mps.is_available():
        return torch.device("mps")
    return torch.device("cpu")

class TabularDataset(Dataset):
    def __init__(self, X: np.ndarray, y: np.ndarray) -> None:
        self.X = torch.as_tensor(X, dtype=torch.float32)
        self.y = torch.as_tensor(y, dtype=torch.float32).unsqueeze(1)

    def __len__(self) -> int:
        return len(self.y)

    def __getitem__(self, idx: int) -> tuple[torch.Tensor, torch.Tensor]:
        return self.X[idx], self.y[idx]

def train_one_epoch(model, loader, criterion, optimizer, device):
    model.train()
    total_loss = 0.0
    for xb, yb in loader:
        xb, yb = xb.to(device), yb.to(device)
        optimizer.zero_grad()                     # 1. Clear stale grads
        logits = model(xb)                        # 2. Forward pass
        loss = criterion(logits, yb)              # 3. Compute loss
        loss.backward()                           # 4. Backprop gradients
        nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()                          # 5. Update weights
        total_loss += loss.item() * len(xb)
    return total_loss / len(loader.dataset)

@torch.inference_mode()
def evaluate(model, loader, criterion, device):
    model.eval()
    total_loss = 0.0
    for xb, yb in loader:
        xb, yb = xb.to(device), yb.to(device)
        loss = criterion(model(xb), yb)
        total_loss += loss.item() * len(xb)
    return total_loss / len(loader.dataset)

A.6.3 Canonical Architecture Blocks (CNN, GRU, Attention, LoRA & Safe Save)

# 1) Compact Medical CNN Block (expects NCHW: (B, C, H, W))
class ConvBlock(nn.Module):
    def __init__(self, c_in: int, c_out: int) -> None:
        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(c_in, c_out, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(c_out),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.net(x)

# 2) Sequence Forecaster (expects (B, T, F) with batch_first=True)
class EpiGRU(nn.Module):
    def __init__(self, n_feat: int, hidden: int = 32) -> None:
        super().__init__()
        self.gru = nn.GRU(n_feat, hidden, batch_first=True)
        self.head = nn.Linear(hidden, 1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        out, _ = self.gru(x)                      # out: (B, T, hidden)
        return self.head(out[:, -1, :])           # last time step

# 3) Multi-Head Attention & LoRA Adapter (Delta W = (alpha / r) * B @ A)
attn = nn.MultiheadAttention(embed_dim=64, num_heads=4, batch_first=True)

class LoRALinear(nn.Module):
    def __init__(self, base: nn.Linear, r: int = 4, alpha: float = 8.0):
        super().__init__()
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False
        self.scale = alpha / r
        self.A = nn.Parameter(torch.randn(r, base.in_features) * 0.02)
        self.B = nn.Parameter(torch.zeros(base.out_features, r))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        delta = (x @ self.A.T) @ self.B.T
        return self.base(x) + self.scale * delta

torch.save(model.state_dict(), "artifacts/best_model.pt")
